 - brew install python@3.12

 - rm -rf venv # 删掉旧的
 - /opt/homebrew/bin/python3.12 -m venv venv # 用3.12创建新的虚拟环境
 - source venv/bin/activate # 激活
 - pip install mcp fastmcp openai requests # 重新安装依赖
